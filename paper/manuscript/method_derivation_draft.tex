\section{Setup}
\label{sec:setup}

A LoRA layer adds the rank-$r$ adapter
$W=BA\in\R^{d_{\mathrm{out}}\times d_{\mathrm{in}}}$ to a frozen weight
($A\in\R^{r\times d_{\mathrm{in}}}$, $B\in\R^{d_{\mathrm{out}}\times r}$). A step
$\Delta A,\Delta B$ changes $W$ to first order by the tangent
$U:=B\Delta A+\Delta B A$ (dropping $\Delta B\,\Delta A$). With
$G=\nabla_W\mathcal L$ and factor gradients $G_A:=B^\top G$, $G_B:=GA^\top$, the
loss decrease splits,
\begin{equation}
  \langle G,U\rangle=\langle G_A,\Delta A\rangle+\langle G_B,\Delta B\rangle.
  \label{eq:split}
\end{equation}

\begin{center}
\begin{tabular}{@{}ll@{}}
$\langle X,Y\rangle=\tr(X^\top Y)$ & Frobenius inner product\\
$\norm{\cdot}_2,\ \norm{\cdot}_*,\ \norm{\cdot}_F$ & spectral, nuclear (dual of spectral), Frobenius norms\\
$\polar(M)=UV^\top$ & for $M=U\Sigma V^\top$ with singular values set to $1$\\
$P,Q\succ0$ & symmetric, Kronecker ansatz $\Sigma=Q\otimes P$ for the gradient 2nd moment (\S\ref{sec:estimate})\\
\end{tabular}
\end{center}

\noindent In the optimizer $G_A,G_B$ are momentum buffers. The derivation uses
raw gradients.

\section{The metric}
\label{sec:metric}

A sample gives a rank-one pseudo-gradient $G_i=b_ix_i^\top$ ($x_i$ input, $b_i$
output sensitivity), disturbing the loss to first order by
$\langle G_i,U\rangle=b_i^\top Ux_i$. The worst single-sample disturbance,
$\max_i|\langle G_i,U\rangle|$, is not enumerable. PoLoRA surrogates it by the
worst over rank-one pseudo-gradients of unit metric size.

\begin{lemma}[Metric Surrogate]
\label{lem:surrogate}
Under the ansatz $\Sigma=Q\otimes P$, with
$\norm{R}_{\Sigma^{-1}}^2:=\vecop(R)^\top\Sigma^{-1}\vecop(R)$,
\begin{equation}
  \Gamma(U):=\sup_{\norm{bx^\top}_{\Sigma^{-1}}\le1}|b^\top Ux|
  =\norm{P^{1/2}UQ^{1/2}}_2 .
  \label{eq:gamma-spectral}
\end{equation}
\end{lemma}

\begin{proof}
The ansatz factors the size,
$\norm{bx^\top}_{\Sigma^{-1}}^2=(b^\top P^{-1}b)(x^\top Q^{-1}x)$. Substituting
$b=P^{1/2}a$, $x=Q^{1/2}z$ makes the objective $a^\top(P^{1/2}UQ^{1/2})z$, with
supremum $\norm{P^{1/2}UQ^{1/2}}_2$ over $\norm{a}\norm{z}\le1$.
\end{proof}

The metric spectral norm $\Gamma$ is thus a surrogate for the worst-sample
disturbance, not the disturbance itself. ($\Sigma^{-1}$ matches the
pseudo-gradient set's shape, Appendix~\ref{app:metric}. The mean-square response
gives the Frobenius surrogate, Appendix~\ref{app:frob}.)

\section{The step: ideal program and surrogate}
\label{sec:program}

PoLoRA is steepest descent in the metric $\Gamma$, split (as in Muon, Scion) into
a direction and a magnitude:
\begin{align}
  \text{direction:}\quad &\argmin_{\Gamma(U)\le1}\langle G,U\rangle,
    \label{eq:ideal}\\
  \text{magnitude:}\quad &\norm{U}_2\le\eta . \nonumber
\end{align}
The LMO is scale-invariant, so the direction step fixes direction only. The trust
region sets the magnitude. The objective \eqref{eq:split} is separable across
factors, but $\Gamma$ is on the joint tangent $U=B\Delta A+\Delta B A$
(Lemma~\ref{lem:surrogate}), so the ideal direction couples $\Delta A$ and
$\Delta B$. Section~\ref{sec:direction} surrogates it, and
Section~\ref{sec:magnitude} sets the magnitude.

\section{Stage one: the direction}
\label{sec:direction}

By subadditivity of the spectral norm,
\begin{equation}
  \norm{P^{1/2}(B\Delta A+\Delta B A)Q^{1/2}}_2
  \le\norm{P^{1/2}B\Delta A\,Q^{1/2}}_2+\norm{P^{1/2}\Delta B\,A\,Q^{1/2}}_2,
  \label{eq:subadd}
\end{equation}
so bounding the two single-factor responses bounds $\Gamma$. PoLoRA surrogates the
joint constraint by these two, descending each factor in its induced metric. Each
whitens to an $r\times r$ form,
\begin{equation}
  \norm{P^{1/2}B\Delta A\,Q^{1/2}}_2=\norm{C_A^{1/2}\Delta A\,Q^{1/2}}_2,
  \qquad
  \norm{P^{1/2}\Delta B\,A\,Q^{1/2}}_2=\norm{P^{1/2}\Delta B\,C_B^{1/2}}_2,
  \label{eq:whiten}
\end{equation}
with $C_A:=B^\top PB$, $C_B:=AQA^\top$ (via $P^{1/2}B=VC_A^{1/2}$, $V^\top V=I$,
orthogonal $V$ preserving the spectral norm). The matrices $C_A,C_B$ are the
factor-gradient covariances (\S\ref{sec:estimate}).

\begin{lemma}[Whitened-Polar LMO]
\label{lem:lmo}
For symmetric positive definite $L,R$ and any $M$, with $H:=L^{-1}MR^{-1}$,
\begin{equation}
  \argmin_{\norm{LXR}_2\le1}\langle M,X\rangle=-L^{-1}\polar(H)R^{-1}.
  \label{eq:lmo}
\end{equation}
\end{lemma}

\begin{proof}
$Y=LXR$ gives $\langle M,X\rangle=\langle H,Y\rangle$ with $\norm{Y}_2\le1$. By
spectral--nuclear duality the minimum is $-\norm{H}_*$, attained at $Y=-\polar(H)$.
\end{proof}

\begin{proposition}[Per-Factor Directions]
\label{prop:dirs}
The objective \eqref{eq:split} under the two per-factor constraints separates
into two LMOs, solved by
\begin{equation}
  W_A=-C_A^{-1/2}\polar(H_A)Q^{-1/2},
  \qquad
  W_B=-P^{-1/2}\polar(H_B)C_B^{-1/2},
  \label{eq:WAWB}
\end{equation}
with $H_A:=C_A^{-1/2}G_AQ^{-1/2}$ and $H_B:=P^{-1/2}G_BC_B^{-1/2}$.
\end{proposition}

\begin{proof}
Apply Lemma~\ref{lem:lmo} with $(L,R,M)=(C_A^{1/2},Q^{1/2},G_A)$ and
$(P^{1/2},C_B^{1/2},G_B)$.
\end{proof}

The oracle is scale-invariant. Keep the unit directions
\[
  \widehat W_A:=W_A/\norm{W_A}_2,\qquad \widehat W_B:=W_B/\norm{W_B}_2.
\]

\section{Stage two: the magnitude}
\label{sec:magnitude}

The directions $\widehat W_A,\widehat W_B$ need a magnitude. A fixed one does not
give a useful step.

\begin{lemma}[Gauge-Unbounded Factor Step]
\label{lem:gauge}
For a fixed $\rho$, the factor step $\Delta A=\rho\widehat W_A$,
$\Delta B=\rho\widehat W_B$ leaves the weight change unbounded over the factor
gauge $(B,A)\mapsto(Bc,A/c)$,
\begin{equation}
  \sup_{c>0}\norm{U}_2=\infty .
  \label{eq:gauge-sup}
\end{equation}
\end{lemma}

\begin{proof}
The gauge fixes $W=BA$ and (via $H_A,H_B$ invariant) the directions
$\widehat W_A,\widehat W_B$, so $U=\rho(c\,B\widehat W_A+c^{-1}\widehat W_B A)$ and
$\norm{U}_2\to\infty$ as $c\to0$ and as $c\to\infty$.
\end{proof}

PoLoRA budgets the weight change directly instead.

\begin{proposition}[Step Magnitude]
\label{prop:length}
The trust region $\norm{U}_2\le\eta$ with the symmetric tie-break
$\norm{\Delta A}_2=\norm{\Delta B}_2=\rho$ gives the step
\begin{equation}
  \Delta A=\rho\,\widehat W_A,\quad
  \Delta B=\rho\,\widehat W_B,\quad
  \rho=\frac{\eta}{\norm{A}_2+\norm{B}_2},
  \label{eq:step}
\end{equation}
which holds $\norm{U}_2\le\eta$ in every gauge.
\end{proposition}

\begin{proof}
The product bound gives
$\norm{U}_2\le\norm{B}_2\norm{\Delta A}_2+\norm{A}_2\norm{\Delta B}_2=(\norm{A}_2+\norm{B}_2)\rho$.
Setting this to $\eta$ gives $\rho$. In the gauge of Lemma~\ref{lem:gauge}
$\rho=\eta/(c\norm{B}_2+c^{-1}\norm{A}_2)$, and the bound stays $\le\eta$.
\end{proof}

A fixed factor step fails because the weight-change size $\norm{U}_2$ is blind to
part of the step. It is zero for a whole family of nonzero factor moves,
$\Delta A=KA$ and $\Delta B=-BK$ with $K\in\R^{r\times r}$, since
$B(KA)-(BK)A=0$, so its ball runs off to infinity along them and cannot pick out a
direction. The metric direction norm has no such blind spot, being zero only at
$\Delta A=\Delta B=0$. So the metric picks the direction and the weight-change
budget only sets its magnitude.

\section{Estimating the metric}
\label{sec:estimate}

$P,Q$ follow from two moment identities of the ansatz, with expectations over
per-example gradients and any compatible $M,N$:
\begin{equation}
  \E[GMG^\top]=\tr(MQ)\,P,
  \qquad
  \E[G^\top NG]=\tr(NP)\,Q .
  \label{eq:moments}
\end{equation}

\begin{lemma}[Moment Closure]
\label{lem:moment}
With $C_A=B^\top PB$ and $C_B=AQA^\top$,
\begin{equation}
  \E\!\left[\tfrac1r G_A^\top C_A^{-1}G_A\right]=Q,
  \qquad
  \E\!\left[\tfrac1r G_BC_B^{-1}G_B^\top\right]=P .
  \label{eq:closure}
\end{equation}
\end{lemma}

\begin{proof}
Apply the second identity of \eqref{eq:moments} with $N=BC_A^{-1}B^\top$. Then
$G^\top NG=G_A^\top C_A^{-1}G_A$ and $\tr(NP)=\tr(I_r)=r$. The other uses
$M=A^\top C_B^{-1}A$.
\end{proof}

PoLoRA keeps the diagonals by EMA:
\begin{equation}
  q_j\leftarrow\mathrm{EMA}\!\left[\tfrac1r(G_A)_{:,j}^\top C_A^{-1}(G_A)_{:,j}\right],
  \qquad
  p_i\leftarrow\mathrm{EMA}\!\left[\tfrac1r(G_B)_{i,:}C_B^{-1}(G_B)_{i,:}^\top\right].
  \label{eq:ema}
\end{equation}
Because $C_A$ depends on $P$ and $C_B$ on $Q$, the optimizer takes one EMA
alternation per step, then freezes $P,Q$ for the step. The inverse square roots
are damped.

\appendix

\section{The metric is the inverse second moment}
\label{app:metric}

Let $g=\vecop(G_i)$ and $\Sigma=\E[gg^\top]\succ0$. The set's extent along a unit
direction $u$ is the root-mean-square component $s(u)=\sqrt{u^\top\Sigma u}$,
built from $\Sigma$ alone and (by polarization) the full second-order shape.

\begin{lemma}[Second-Moment Ellipsoid]
\label{lem:shape}
Among centered ellipsoids $E_A=\{g:g^\top Ag\le1\}$, $A\succ0$, the unique one
whose support width matches the extent in every direction,
$\max_{g\in E_A}u^\top g=s(u)$ for all $u$, has $A=\Sigma^{-1}$.
\end{lemma}

\begin{proof}
The support width of $E_A$ is $\sqrt{u^\top A^{-1}u}$. Equating to
$s(u)=\sqrt{u^\top\Sigma u}$ for all $u$ gives $A^{-1}=\Sigma$.
\end{proof}

So the unit ball of $\norm{\cdot}_{\Sigma^{-1}}$ reaches, in each direction, as
far as the pseudo-gradients spread.

\section{Mean-square response: the Frobenius surrogate}
\label{app:frob}

\begin{lemma}[Mean-Square Response]
\label{lem:frob}
Under the ansatz $\Sigma=Q\otimes P$, the average squared sample interaction is
\begin{equation}
  \E_i[\langle G_i,U\rangle^2]=\vecop(U)^\top\Sigma\,\vecop(U)=\norm{P^{1/2}UQ^{1/2}}_F^2 .
  \label{eq:frob-response}
\end{equation}
\end{lemma}

Its LMO replaces the spectral--nuclear duality of Lemma~\ref{lem:lmo} by
Cauchy--Schwarz ($\polar(H)\to H/\norm{H}_F$), giving the per-factor directions
$W_A^{\mathrm F}\propto C_A^{-1}G_AQ^{-1}$ and $W_B^{\mathrm F}\propto P^{-1}G_BC_B^{-1}$.
Worst-case over samples surrogates the disturbance by the spectral polar
direction. Mean-square surrogates it by the Frobenius inverse direction.
