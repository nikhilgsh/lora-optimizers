\section{Using Spectral Geometry with Lora}

Consider a single linear layer, where $W_0 \in \R^{d_{\mathrm{out}} \times d_{\mathrm{in}}}$ is the frozen pre-trained weights of this layer. Our objectives is to fine-tune a given model (with frozen weights) for a given task by adding a low rank matrix to these weights. We will represent this fine-tuning task as
\begin{equation}
    \min_{A\in\R^{r\times d_{\mathrm{in}}},
B\in\R^{d_{\mathrm{out}}\times r}} \ell(W_0 + BA)
\end{equation}
where $\ell$ is a loss that captures this task.  The resulting linear layers of our model are now encoded by
\[ x \mapsto W_0 \,x + BA \,x.\]

\section{Method: \methodname}
\label{sec:method}
% MAIN STORY (outlined): the per-step DIRECTION of all three is the linear-minimization
% oracle (LMO) of the gradient under a norm -- 3.1 spectral norm per factor, 3.2 the
% decoupled half-split product step, 3.3 learned two-sided diagonal-curvature metric. OURS IS NOT A PURE LMO:
% the metric P,Q is a coupled diagonal KL flip-flop (estimation, not an LMO), and the
% MAGNITUDE is the operator-norm pin rho=eta/(smax A+smax B) (a separate rule, not the LMO
% budget tau). Keep those three ingredients distinct in the prose.

% Citation keys (verify in references.bib): kang_specgf, imuon, spectron,
% polarexpress, soap, mua, tilde_csd (compositional steepest descent -- no arXiv).

% --- setup + the steepest-descent (LMO) view (MAIN STORY) ---------------------
% All three updates are the same step (steepest descent under a norm) under
% different norms. Statements in body; proofs in \Cref{app:proofs}.
\paragraph{Setup.}
\begin{itemize}[leftmargin=1.5em,itemsep=1pt,topsep=2pt]
  \item LoRA factors $A\in\R^{r\times d_{\mathrm{in}}}$, $B\in\R^{d_{\mathrm{out}}\times r}$; adapter $\DW=(\alpha/r)\,BA$.
  \item $G\triangleq\partial L/\partial(BA)$: gradient of the loss w.r.t.\ the \emph{unscaled} product $BA$ (the $\alpha/r$ scale is absorbed into the learning rate $\eta>0$). Per-factor gradients $G_A=B^\top G$, $G_B=GA^\top$.
  \item $\dot A,\dot B$: the updates applied to the factors ($A\leftarrow A+\dot A$).
  \item $\polar(M)=UV^\top$ for the SVD $M=U\Sigma V^\top$ --- the \emph{matrix sign} of $M$ (it sets each nonzero singular value to $1$; the orthogonal polar factor); spectral norm $\norm{M}_2=\smax(M)$; $\langle X,Y\rangle=\operatorname{tr}(X^\top Y)$; Gram matrices $S_A=AA^\top$, $S_B=B^\top B$.
\end{itemize}
\textbf{Steepest descent under a norm} --- the linear minimization oracle (LMO) --- for a factor gradient $M$ and a norm $\norm{\cdot}$, is the update
\begin{equation}
  \eta\,\argmin_{\norm{U}\le1}\langle M,U\rangle .
  \label{eq:lmo}
\end{equation}
The three variants specialize $\norm{\cdot}$ to three different norms; \Cref{lem:lmo} is the case of the spectral norm $\norm{\cdot}_2$.

\subsection{Naive Muon on the factors}
\label{sec:method-naive}
\paragraph{Norm.} The spectral norm $\norm{\cdot}_2$, applied to each factor separately.
\begin{lemma}[Spectral-norm LMO]\label{lem:lmo}
$-\polar(M)\in\argmin_{\norm{U}_2\le1}\langle M,U\rangle$ for any matrix $M$ (uniquely so when $M$ has full rank with distinct singular values). \emph{(\Cref{app:proofs}.)}
\end{lemma}
\noindent Applying \Cref{lem:lmo} to each factor gradient:
\begin{equation}
  \dot A=-\eta\,\polar(G_A),\qquad \dot B=-\eta\,\polar(G_B).
  \label{eq:naive}
\end{equation}
Two defects on a factored weight:
\begin{itemize}[leftmargin=1.5em,itemsep=1pt,topsep=2pt]
  \item \emph{Factorization-dependent.} The product $\DW=BA$ is invariant under $(A,B)\mapsto(NA,BN^{-1})$ for invertible $N$, but the factor gradient transforms as $G_A\mapsto N^{-\top}G_A$. Since $\polar$ is not equivariant, \eqref{eq:naive} depends on the factorization, not on $\DW$ alone \cite{imuon}.
  \item \emph{Unbounded merged step.} The merged step
        $\norm{B\dot A+\dot B A}_2\le\eta\big(\norm{A}_2+\norm{B}_2\big)$
        grows with the factor norms, which can diverge under the corresponding flow \cite{kang_specgf}.
\end{itemize}

\subsection{iMuon / compositional Muon}
\label{sec:method-imuon}
\paragraph{LMO (half-split).} Budget each factor's contribution to the product step separately --- the tractable surrogate \cite{tilde_csd} for the harder joint constraint $\norm{\dot B A+B\dot A}_2\le\eta$:
\begin{equation}
  \min_{\dot A}\ \langle G_A,\dot A\rangle \ \ \text{s.t.}\ \ \norm{B\dot A}_2\le\eta,
  \qquad
  \min_{\dot B}\ \langle G_B,\dot B\rangle \ \ \text{s.t.}\ \ \norm{\dot B A}_2\le\eta .
  \label{eq:imuon-lmo}
\end{equation}
\begin{proposition}[Decoupled factor step]\label{prop:imuon}
The solution of \eqref{eq:imuon-lmo} whitens each factor gradient by the other factor's Gram root:
\begin{equation*}
  \dot A=-\eta\,S_B^{-1/2}\,\polar\!\big(S_B^{-1/2}G_A\big),\qquad
  \dot B=-\eta\,\polar\!\big(G_B S_A^{-1/2}\big)\,S_A^{-1/2}.
\end{equation*}
\emph{(\Cref{app:proofs}; iMuon \cite{imuon}, and the half-split of compositional Muon \cite{tilde_csd}.)}
\end{proposition}
\begin{corollary}[Product-step invariance]\label{cor:imuon}
$\norm{B\dot A}_2=\eta$, independent of the factor norms --- so no runtime rescale is needed. \emph{(\Cref{app:proofs}.)}
\end{corollary}
\begin{itemize}[leftmargin=1.5em,itemsep=1pt,topsep=2pt]
  \item \emph{Same operator, other compositions.} Compositional Muon applies it to the attention products $W_QW_K^\top$ and $W_OW_V$ \cite{tilde_csd}.
  \item \emph{Same operator, three independent incarnations.} The memoryless limit of LoRA-RITE \cite{lorarite} (moment accumulation stripped), iMuon's decoupled form \cite{imuon}, and the concurrent LoRA-Muon \cite{loramuon} all realize \Cref{prop:imuon} (\Cref{app:memoryless}); \cite{loramuon} proves this equivalence. Our ablation runs this shared update's \emph{direction} as the both-controls-removed arm, with the step magnitude pinned by the unit-spectral-norm rescale of \Cref{alg:ours} (\Cref{sec:exp}); the unpinned update's factor step is uncapped ($\propto1/\sigma_{\min}$ of the partner).
  \item \emph{The gap we close.} The raw factor step is uncontrolled --- $\norm{\dot A}_2$ scales as $\eta/\sigma_{\min}(B)$ --- with no per-factor operator-norm radius to bound it.
\end{itemize}

\subsection{Ours: \methodname}
\label{sec:method-ours}
\paragraph{Motivation.} Naive Muon (\Cref{lem:lmo}) and iMuon (\Cref{prop:imuon}) are the same spectral LMO under the Euclidean inner product, differing only in what they cap. \methodname{} keeps a spectral LMO for the \emph{direction} but measures it in a learned two-sided metric, and sets the \emph{magnitude} by a separate rule. Three stages:
\begin{itemize}[leftmargin=1.7em,itemsep=1pt,topsep=2pt]
  \item[(A)] \emph{Metric.} Estimate diagonal $P,Q$ online as the Kronecker factors of the gradient second moment (coupled diagonal second-moment matching \cite{klshampoo}; \Cref{par:coupling}).
  \item[(B)] \emph{Direction.} Spectral-norm LMO of the loss in that metric --- a whiten--polar--unwhiten step (\Cref{prop:ours}).
  \item[(C)] \emph{Magnitude.} Discard the LMO budget; re-pin to the operator-norm radius (\Cref{eq:ours-prodcap}).
\end{itemize}
Stages (A) and (B) are separate problems --- estimation, then a step under the \emph{frozen} estimate --- composed by alternation, not one objective.

\paragraph{(A) The metric.} Model the gradient second moment as diagonal--Kronecker:
\begin{equation}
  \Sigma:=\E[\mathrm{vec}(G)\,\mathrm{vec}(G)^\top]\approx Q\otimes P,\qquad
  P=\diag(p)\succ0\ (d_{\mathrm{out}}),\quad Q=\diag(q)\succ0\ (d_{\mathrm{in}}),
  \label{eq:ours-sigma}
\end{equation}
the Shampoo Kronecker factorization \cite{soap} under an Adafactor diagonal restriction \cite{adafactor}. Whitening by $\Sigma^{-1/2}=Q^{-1/2}\otimes P^{-1/2}$ sends $\mathrm{vec}(G)\mapsto\mathrm{vec}(P^{-1/2}GQ^{-1/2})$ --- the matrix analogue of Adam's $1/\!\sqrt{v}$, one square root per Kronecker side. Whitening by this $-1/2$ power of the second moment is the covariance / optimal-Kronecker choice \cite{morwani,klshampoo}, equivalently Adafactor in Shampoo's eigenbasis \cite{soap}; original Shampoo uses the $-1/4$ power \cite{shampoo} --- half the whitening, from a Loewner-bound rather than a second-moment motivation. We take the spectral (operator-norm) cap in this whitened geometry, measuring the product step in
\begin{equation}
  \norm{X}_{P,Q}=\norm{P^{1/2}XQ^{1/2}}_2,
  \label{eq:ours-metric}
\end{equation}
the operator norm of $X$ after the $\Sigma^{1/2}=Q^{1/2}\otimes P^{1/2}$ change of variables ($P^{1/2}XQ^{1/2}$ is $\Sigma^{1/2}$ applied to $X$). $P=Q=I$ recovers the Euclidean spectral case of \Cref{sec:method-naive,sec:method-imuon}; the diagonals are estimated online (decay $\beta_2$) by \eqref{eq:ours-klcoupling}.

\paragraph{Estimating the metric.}
\label{par:coupling}
We never form $G$. The factor gradients $G_A=B^\top G$ and $G_B=GA^\top$ are linear images of it, so their row second moments are $P,Q$ conjugated by the partner factor (\Cref{app:matnormal}):
\begin{equation}
  C_A=B^\top P B,\qquad C_B=A\,Q\,A^\top
  \label{eq:ours-closure}
\end{equation}
($r\times r$, formed from the diagonals on the fly, never stored). Matching the diagonal of each factor gradient's second moment to the model \eqref{eq:ours-sigma} ($G_A$ resolves $Q$, $G_B$ resolves $P$) gives the per-coordinate estimator:
\begin{equation}
  q_j=\mathrm{EMA}\Big[\tfrac1r\,(G_A)_{:,j}^\top C_A^{-1}(G_A)_{:,j}\Big],\qquad
  p_i=\mathrm{EMA}\Big[\tfrac1r\,(G_B)_{i,:}\,C_B^{-1}(G_B)_{i,:}^\top\Big].
  \label{eq:ours-klcoupling}
\end{equation}
It is unbiased ($\E[\tfrac1r\,G_A^\top C_A^{-1}G_A]=Q$, \Cref{app:matnormal}) and couples the sides through $C_A,C_B$ ($C_A$ injects $P$, $C_B$ injects $Q$); $C_A=C_B=I$ drops to two independent Adafactor diagonals \cite{adafactor}. This is the coupled diagonal second-moment matching of \cite{klshampoo}, specialised to the LoRA factor gradients; we run one $\beta_2$-EMA alternation per step.

\paragraph{(B) The step: spectral LMO in the frozen metric.} With $P,Q$ \emph{frozen} at their current estimate, cap the product step in the metric \eqref{eq:ours-metric} and maximize alignment with the gradient:
\begin{equation}
  \max_{\dot A}\ \langle B\dot A,\,-G\rangle
  \quad\text{s.t.}\quad
  \norm{B\dot A}_{P,Q}\le\tau .
  \label{eq:ours-opt}
\end{equation}
This is the spectral LMO of \Cref{lem:lmo} in the $P,Q$ metric, with $P,Q$ from (A) held fixed. Whitening both sides by $P^{-1/2},Q^{-1/2}$ reduces it to the Euclidean spectral case, giving a closed form.
\begin{proposition}[Metric LMO]\label{prop:ours}
For $B$ of full column rank, the solution of \eqref{eq:ours-opt} is
\begin{equation*}
  \dot A = -\tau\,C_A^{-1/2}\,\polar\!\big(C_A^{-1/2}\,G_A\,Q^{-1/2}\big)\,Q^{-1/2},
  \qquad C_A=B^\top P B .
\end{equation*}
\end{proposition}
\medskip
\noindent \methodname{} uses \Cref{prop:ours} for the step \emph{direction} only, on the bias-corrected Nesterov momentum $\hat M_A$ in place of the gradient:
\begin{equation}
  W_A \;=\; C_A^{-1/2}\,\polar\!\big(C_A^{-1/2}\,\hat M_A\,Q^{-1/2}\big)\,Q^{-1/2},
  \label{eq:ours-lmo}
\end{equation}
with $W_B$ the mirror ($C_B=AQA^\top$).

\paragraph{(C) The magnitude: operator-norm radius.} Since $\polar(\cdot)$ has unit spectral norm, the magnitude of $W_A$ is an artifact of the whitening, not the LMO budget $\tau$ --- stage (B) fixes only the \emph{direction}. \Cref{alg:ours} \emph{discards} it and re-pins the step to the operator-norm radius $\rho=\eta/(\norm{A}_2+\norm{B}_2)$ via $\dot A=-\rho\,W_A/\norm{W_A}_2$, which caps the merged step independent of conditioning:
\begin{equation}
  \norm{B\dot A+\dot B A}_2\le\rho\big(\norm{A}_2+\norm{B}_2\big)=\eta .
  \label{eq:ours-prodcap}
\end{equation}
This is a second trust region, separate from (B): a Euclidean operator-norm cap on the merged step, not the $P,Q$ cap of \eqref{eq:ours-opt} (whose budget $\tau$ is discarded). It makes the method stable from the standard $B{=}0$ init (\Cref{rem:b0}).

\begin{remark}[Relation to AdaPreLoRA]
\Cref{prop:ours} caps the whitened step in operator norm, giving the polar; a Frobenius cap instead gives the linear solve $\dot A\propto C_A^{-1}G_A Q^{-1}$ --- AdaPreLoRA \cite{adaprelora}, the same projection without the spectral cap.
\end{remark}

\begin{algorithm}[t]
\caption{\methodname{} step for one LoRA pair $(A,B)$ at iteration $t$}
\label{alg:ours}
\begin{algorithmic}[1]
\Require gradients $G_A,G_B$; momenta $M_A,M_B$; diagonals $P,Q$; decays $\beta_1,\beta_2$; rate $\eta$
\Phase{Momentum (Nesterov, bias-corrected)}
\State $M_A\gets\beta_1 M_A+(1-\beta_1)G_A$;\quad $\hat M_A\gets\big(\beta_1 M_A+(1-\beta_1)G_A\big)/(1-\beta_1^{t})$ \Comment{mirror for $B$}
\Phase{Whiten, polar, unwhiten}
\State $C_A\gets B^\top P B$;\quad $C_B\gets A Q A^\top$ \Comment{$r\times r$}
\State $W_A\gets C_A^{-1/2}\,\polar\!\big(C_A^{-1/2}\hat M_A Q^{-1/2}\big)\,Q^{-1/2}$
\State $W_B\gets P^{-1/2}\,\polar\!\big(P^{-1/2}\hat M_B C_B^{-1/2}\big)\,C_B^{-1/2}$
\Phase{Operator-norm step}
\State $\rho\gets\eta/(\norm{A}_2+\norm{B}_2)$
\State $A\gets A-\rho\,W_A/\norm{W_A}_2$;\quad $B\gets B-\rho\,W_B/\norm{W_B}_2$
\Phase{Metric update (flip-flop, one alternation)}
\State $Q_{jj}\gets\beta_2 Q_{jj}+\tfrac{1-\beta_2}{r}(G_A)_{:,j}^\top C_A^{-1}(G_A)_{:,j}$
\State $P_{ii}\gets\beta_2 P_{ii}+\tfrac{1-\beta_2}{r}(G_B)_{i,:}C_B^{-1}(G_B)_{i,:}^\top$
\end{algorithmic}
\end{algorithm}
\noindent Every $\norm{\cdot}_2$ above is the guarded $\smax$ estimate of \Cref{alg:smax}, and every inverse-sqrt is relative-damped (\Cref{app:damping}).

\begin{remark}[Why the operator-norm step is safe at the standard $B{=}0$ initialization]
\label{rem:b0}
LoRA starts at $B{=}0$, where the decoupled factor step of \Cref{prop:imuon} has no stable learning rate but the operator-norm step does.
\begin{itemize}[leftmargin=1.5em,itemsep=1pt,topsep=2pt]
  \item \textbf{The failure.} $\polar(S_B^{-1/2}G_A)$ has unit spectral norm however small $G_A$ is, and the outer $S_B^{-1/2}$ scales it by up to $1/\sigma_{\min}(B)$, so $\norm{\dot A}_2\approx\eta/\sigma_{\min}(B)$ --- unbounded as $B\to0$, undefined at $B{=}0$. Renormalizing to a scale-free step removes the gradient's natural $\norm{G_A}\propto\sigma(B)$ protection; lowering $\eta$ does not help (it slows $B$'s escape by the same factor it shrinks the spurious step). So the decoupled spectral update --- iMuon's closed form (Cor.\ 4.1), re-derived by LoRA-Muon \cite{loramuon}, equivalent to simplified LoRA-RITE \cite{lorarite} --- has no stable rate from this init (\Cref{sec:exp}).
  \item \textbf{The family's workarounds.} iMuon applies the metric LMO to a \emph{joint} product-space momentum $M=\dot M_BA+B\dot M_A$ (a different update, not the decoupled solution) and prepends warmup to grow $B$ off zero; LoRA-Muon trains from nonzero random factors and never enters the regime.
  \item \textbf{Ours needs neither.} The operator-norm step $\dot A=-\rho\,W_A/\norm{W_A}_2$ pins $\norm{\dot A}_2=\rho$ at every conditioning; at $B{=}0$, $\rho=\eta/(\norm{A}_2+\norm{B}_2)=\eta/\norm{A}_2$ with $A$ a full-scale random factor --- an ordinary step. The polar discards the gradient's magnitude and the rescale re-imposes the radius, so the step size is a property of the method, not the factor conditioning.
\end{itemize}
\end{remark}

\paragraph{Cost.} \methodname{} is \emph{lighter} than AdamW in memory and adds a small, batch-amortized compute overhead.
\begin{itemize}[leftmargin=1.5em,itemsep=1pt,topsep=2pt]
  \item \textbf{Memory.} Same first moment as AdamW, but AdamW's second moment is replaced by two curvature diagonals $P,Q$ --- a factor $r$ smaller. The $r\times r$ matrices $C_A=B^\top PB$, $C_B=AQA^\top$ are formed on the fly (\Cref{alg:ours}), not stored, and the inverse-sqrt is eigh-free (no eigenbasis, no $d\times d$ state). Net state ${\approx}\tfrac12$ AdamW's (\Cref{tab:optstate}).
  \item \textbf{Compute.} Per pair: $8$ matmuls of cost ${\sim}r^2d$ (forming $C_A,C_B$, whiten/un-whiten, two curvature applies), two full polars (PolarExpress \cite{polarexpress}, $K{=}8$ steps), two gram--Newton--Schulz inverse-sqrts \cite{gram_ns} (${\sim}r^3$). All $r\times r$ or $r\times d$, eigh-free; the polars dominate.
  \item \textbf{Wall.} Near-free at production batch: the fixed \texttt{optimizer.step} amortizes against the batch-scaling forward/backward (\Cref{sec:exp-walltime}).
\end{itemize}

\section{Prior work}
\paragraph{LoRa and optimization of low rank factors.}
LoRA finetuning freezes the pretrained weight matrix and trains a low-rank update
$\DW=(\alpha/r)BA$ through the factors $A$ and $B$ \cite{lora}.
AdamW is a standard optimizer choice applied independently to the two
factors \cite{adamw}.  
This is a simple and memory efficient strategy, however it introduces a problem since it
optimizes the parameterization rather than the product.
Specifically, the same update
$\DW$ can be represented by $(A,B)$ or by $(NA,BN^{-1})$, while a
factor-wise adaptive optimizer need not give the same update to $\DW$.
Several prior work address this issue indirectly by changing
the parameterization, initialization, or scalar learning rates.  
LoRA+
uses different learning rates for $A$ and $B$ \cite{loraplus};
rank-stabilized LoRA changes the $\alpha/r$ scaling to improve
high-rank behavior \cite{rslora}; $\mu$A and recent LoRA scaling rules
study how learning rates and $\alpha$ should transfer across rank
\cite{mua,loraalpha}.  These works change the scale at which the
factor-based optimizer operates, while \methodname{} keeps the usual
LoRA parameterization and changes the update direction and radius.


\paragraph{Orthogonalized updates.}
Muon~\cite{muon} update is derived as a steepest descent over the spectral norm
ball, in other words spectral descent.  
The spectral descent direction gave rise to a line of work on
spectrally controlled training.  
When Muon is applied separately to LoRa factors, spectral gradient flow exhibits a distinctive spectral phenomenon, 
``equal-rate'' growth of singular values, explaining why factor-wise
orthogonalization can behave very differently from ordinary gradient
flow \cite{kang_specgf}.  Spectron instead uses orthogonalization and
adaptive renormalization to stabilize native low-rank pretraining by
controlling the spectral norm of the induced weight update
\cite{spectron}.  These methods motivate spectral update, but they do
not by themselves provide a product-aware optimizer.  \methodname{} uses the same polar
primitive, but applies it inside a LoRA-aware curvature metric and
then rescales the resulting factor step by a product-level operator-norm
radius.


\paragraph{Curvature and second-moment preconditioning.}
AdamW and Adafactor use diagonal or factored second moments
\cite{adamw,adafactor}.  Shampoo extends this idea to matrix and tensor
parameters by approximating full matrix Adagrad with Kronecker-factored
matrix
\cite{shampoo}; SOAP improves Shampoo by applying Adam in the
Shampoo preconditioner eigenbasis \cite{soap}.  
Further, a recent KL perspective
interprets Shampoo and SOAP as structured covariance-estimation
procedures and motivates KL-Shampoo/KL-SOAP \cite{klshampoo}, which is
related to our metric estimator.

Curvature-aware spectral optimizers combine this second-moment geometry
with Muon-style orthogonalization.  SOAP-Muon first whitens with a SOAP
preconditioner and then applies Muon \cite{soapmuon}.  Mousse~ \cite{mousse} formulates
a dense full-rank update as a spectral steepest descent step in a
Shampoo-induced anisotropic trust region,
\[
    \Delta W
    =
    -L^{-1/4}
      \polar\left(L^{-1/4} G R^{-1/4}\right)
      R^{-1/4},
\]
where $L$ and $R$ are Kronecker second-moment factors.
\methodname{} is the LoRA member of this family.  Its whiten-polar-
unwhiten step has the same form, but the curvature seen by a LoRA pair
collapses to the small matrices
\[
    C_A=B^\top P B,\qquad C_B=AQA^\top ,
\]
together with coupled diagonal side factors $P$ and $Q$.  Thus it keeps the
spectral-curvature direction, while avoiding dense $d\times d$
preconditioners and their eigenbases.




\paragraph{Product-space and invariant LoRA optimizers.}
A LoRA adapter is invariant to the change of coordinates
\[
    BA=(BN^{-1})(NA),
\]
for any invertible matrix $N\in \mathbb{R}^{r\times r}$,
so an optimizer should ideally act on the product $BA$, not on the
individual factors.  
LoRA-RITE \cite{lorarite} is prior work that preconditions only the
shared rank-$r$ coordinate of the LoRA factors, rather than the full
input/output dimensions.  This rank-side preconditioning cancels the
change of basis $(A,B)\mapsto(NA,BN^{-1})$, making the induced update to
$BA$ transformation-invariant.
Riemannion/RiemannLoRA \cite{riemannion} is also prior work
that optimizes the adapter directly on the fixed-rank manifold.  The
recent concurrent works iMuon \cite{imuon} and LoRA-Muon
\cite{loramuon} derive the spectral steepest descent version: iMuon as a
general intrinsic LMO framework for matrix manifolds, and LoRA-Muon as
the LoRA-specialized form.

In the memoryless spectral limit, these methods reduce to the same update
\[
  \dot A
  =
  -\eta\,S_B^{-1/2}\,
  \polar\!\left(S_B^{-1/2}G_A\right),
  \qquad
  \dot B
  =
  -\eta\,
  \polar\!\left(G_BS_A^{-1/2}\right)S_A^{-1/2},
\]
with $S_A=AA^\top$, $S_B=B^\top B$, $G_A=B^\top G$, and
$G_B=GA^\top$.  This update is invariant at the product level and caps
the two half-product steps.  It does not, however, cap the raw factor
steps, which scale as $1/\sigma_{\min}$ of the partner factor and are
therefore singular at the standard $B=0$ LoRA initialization.
\methodname{} uses this shared spectral core as the invariant baseline,
then adds a coupled two-sided curvature metric for the direction and a
separate operator-norm radius for the merged adapter step.

\paragraph{Curvature-whitened spectral updates.} Measuring the spectral (Muon) cap in a Shampoo-curvature metric has a dense, full-rank precedent in pre-training: SOAP-Muon \cite{soapmuon} (one-sided SOAP whitens the small side, Muon orthogonalizes the other, converging to a two-sided equilibrium) and, closest to us, Mousse \cite{mousse}. With one-sided second-moment factors $L=\mathrm{EMA}(GG^\top)$ and $R=\mathrm{EMA}(G^\top G)$, Mousse takes the step
\begin{equation*}
  \DW=-L^{-1/4}\,\polar\!\big(L^{-1/4}\,G\,R^{-1/4}\big)\,R^{-1/4},
\end{equation*}
the same whiten/polar/unwhiten sandwich as \eqref{eq:ours-lmo}, but at the quarter power $L^{-1/4}$ \cite{shampoo} (tempered to $1/8$ in practice) where we whiten fully at $\Sigma^{-1/2}$. \methodname{} differs on four further axes that matter for adapter fine-tuning:
\begin{itemize}[leftmargin=1.5em,itemsep=1pt,topsep=2pt]
  \item \textbf{Factor space.} On a LoRA pair the curvature never forms a dense $d\times d$ matrix. It collapses to two small $r\times r$ matrices, $C_A=B^\top PB$ and $C_B=AQA^\top$ --- the curvature seen on the rank-$r$ subspaces the factors actually span --- in place of the dense Kronecker factors of Mousse/SOAP-Muon.
  \item \textbf{Diagonal but coupled.} The two side-factors $P,Q$ are diagonal (Adafactor-form \cite{adafactor}) but coupled through the second-moment matching of \cite{klshampoo}: not independent per-axis gradient energies, and not full Shampoo eigenbases. Our ablations show this dense, full-eigenbasis flavor does not improve on the coupled diagonal.
  \item \textbf{LoRA-native.} It is the adapter member of this family, where the dense methods target full-parameter pre-training.
  \item \textbf{Magnitude without grafting.} Mousse's unwhitened step $L^{-1/4}(\cdot)R^{-1/4}$ shrinks as the curvature factors $L,R$ grow, so it grafts the step to the whitened update's Frobenius norm to hold the update scale across training \cite{mousse}; our radius $\rho=\eta/(\norm{A}_2{+}\norm{B}_2)$ \eqref{eq:ours-prodcap} fixes $\norm{B\dot A+\dot B A}_2=\eta$ from the current factors rather than the curvature, so the scale cannot drift and grafting is unnecessary.
\end{itemize}

\paragraph{Norm-scaled per-factor learning rates.} QuacK \cite{quack} (concurrent) shares our aim of keeping a \emph{product} of two trained matrices from moving too far in one step, but reaches it by rescaling learning rates rather than by capping the step. On the attention-logit product $W_QW_K^\top$ it gives each weight a learning rate equal to a base rate $\eta_0$ over the Frobenius norm of its partner,
\begin{equation}
  \eta^{(Q)}=\frac{\eta_0}{\norm{W_K}_F},\qquad
  \eta^{(K)}=\frac{\eta_0}{\norm{W_Q}_F},
  \label{eq:quack}
\end{equation}
so a larger $W_K$ slows $W_Q$ and vice versa. Our radius \eqref{eq:ours-prodcap} does the same job on the adapter product $BA$, through a single scalar $\rho=\eta/(\norm{A}_2{+}\norm{B}_2)$ that holds the merged step at $\norm{B\dot A+\dot B A}_2=\eta$. Three differences:
\begin{itemize}[leftmargin=1.5em,itemsep=1pt,topsep=2pt]
  \item \textbf{What moves:} for QuacK, the attention logits during pre-training \eqref{eq:quack}; for us, a fine-tuning adapter.
  \item \textbf{Which norm:} QuacK divides by the Frobenius norm of the partner factor (they report the spectral norm performs comparably); we divide by the operator norm, and cap the operator norm of the merged step.
  \item \textbf{What the rate multiplies:} QuacK rescales the base-optimizer step on each factor, leaving its direction unchanged; our $\rho$ rescales the polar direction \eqref{eq:ours-lmo} --- the curvature-whitened, orthogonalized update --- so it bounds the magnitude on top of a changed direction.
\end{itemize}

\section{Proofs}
\label{app:proofs}

\begin{proof}[Proof of \Cref{lem:lmo}]
Let $M=U\Sigma V^\top$. For any $U'$ with $\norm{U'}_2\le1$,
\[
  \langle M,U'\rangle=\sum_i\sigma_i\,u_i^\top U' v_i \;\ge\; -\sum_i\sigma_i,
\]
since $|u_i^\top U' v_i|\le\norm{U'}_2\le1$. Equality holds at $U'=-\polar(M)$, uniquely when $M$ has full rank with distinct singular values; the zero and repeated singular directions otherwise leave $U'$ free.
\end{proof}

\begin{proof}[Proof of \Cref{prop:imuon}]
The objective separates over the two factors:
\[
  \langle G,\ \dot B A+B\dot A\rangle=\langle G_A,\dot A\rangle+\langle G_B,\dot B\rangle .
\]
Take the $A$-block under its budget $\norm{B\dot A}_2\le\eta$. Write $B=U_B S_B^{1/2}$ with $U_B^\top U_B=I$, so that $\norm{B\dot A}_2=\norm{S_B^{1/2}\dot A}_2$. The substitution $Z=S_B^{1/2}\dot A$ turns the block into
\[
  \min_{\norm{Z}_2\le\eta}\ \langle S_B^{-1/2}G_A,\,Z\rangle,
\]
which \Cref{lem:lmo} solves at $Z=-\eta\,\polar(S_B^{-1/2}G_A)$. Undoing the substitution gives
\[
  \dot A=S_B^{-1/2}Z=-\eta\,S_B^{-1/2}\polar\!\big(S_B^{-1/2}G_A\big),
\]
and the $B$-block is symmetric. This recovers iMuon's idealized (decoupled) update \cite{imuon}, the per-factor split of compositional Muon \cite{tilde_csd}.
\end{proof}

\begin{proof}[Proof of \Cref{cor:imuon}]
With $U_B=BS_B^{-1/2}$ (orthonormal columns),
\[
  B\dot A=-\eta\,U_B\,\polar(S_B^{-1/2}G_A).
\]
Left-multiplication by $U_B$ preserves the spectral norm, and $\polar(\cdot)$ has unit nonzero singular values, so $\norm{B\dot A}_2=\eta$.
\end{proof}

\begin{proof}[Proof of \Cref{prop:ours}]
Write $C_A=B^\top P B$ and substitute $Y_A=C_A^{1/2}\dot A\,Q^{1/2}$, so $\dot A=C_A^{-1/2}Y_A Q^{-1/2}$. Since $P^{1/2}B\,C_A^{-1/2}$ has orthonormal columns (its Gram is $I$), left-multiplication preserves singular values:
\[
  \norm{B\dot A}_{P,Q}=\norm{P^{1/2}B\dot A\,Q^{1/2}}_2=\norm{Y_A}_2 .
\]
The objective is plain in $Y_A$ as well: with $G_A=B^\top G$ and trace cyclicity,
\[
  \langle B\dot A,-G\rangle=\langle\dot A,-G_A\rangle=\langle Y_A,-H_A\rangle,\qquad H_A=C_A^{-1/2}G_A Q^{-1/2}.
\]
So the problem reduces to $\max_{\norm{Y_A}_2\le\tau}\langle Y_A,-H_A\rangle$, solved by $Y_A=-\tau\,\polar(H_A)$ (\Cref{lem:lmo}); unwhitening,
\[
  \dot A=-\tau\,C_A^{-1/2}\,\polar\!\big(C_A^{-1/2}G_A Q^{-1/2}\big)\,Q^{-1/2}.
\]
The Frobenius ball $\norm{Y_A}_F\le\tau$ instead gives $\dot A\propto C_A^{-1}G_A Q^{-1}$, recovering AdaPreLoRA.
\end{proof}

\section{Memoryless limits of LoRA-RITE and iMuon}
\label{app:memoryless}
LoRA-RITE \cite{lorarite} and iMuon \cite{imuon} both reduce to the decoupled factor step of
\Cref{prop:imuon} once their moment accumulation is removed. We show this below. The one
caveat is iMuon: it is the \emph{momentum-free} algorithm that collapses, while the shipped
momentum variant does not (last paragraph) --- the code-versus-theory gap of
\Cref{rem:b0,sec:related}.

\paragraph{Setup.} As in \Cref{sec:method}: $G=\partial L/\partial(BA)$ is the product
gradient, $G_A=B^\top G$ and $G_B=GA^\top$ the factor gradients, $S_A=AA^\top$ and
$S_B=B^\top B$ the $r\times r$ Grams. Both derivations use one fact --- the first half of a
matrix's polar decomposition is its polar factor: for full-column-rank $M=U\Sigma V^\top$,
\begin{equation}
  M\,(M^\top M)^{-1/2}=UV^\top=\polar(M).
  \label{eq:polar-id}
\end{equation}

\paragraph{LoRA-RITE.} (LoRA-RITE writes the two factors in the opposite order; we state its
update directly for our $A,B$.) Its Alg.~1 adds three pieces of memory to a base step:
\begin{itemize}[leftmargin=1.5em,itemsep=1pt,topsep=2pt]
  \item an EMA first moment (decay $\beta_1$);
  \item an accumulated second moment, transported across the (changing) factor basis each step;
  \item an ``escaped-mass'' floor added to that second moment.
\end{itemize}
Dropping all three --- $\beta_1{=}0$, and the second moment taken as the single-step value
with no transport and no floor --- is the authors' momentum-free update (their Eq.~(14)),
which in our notation reads
\begin{equation*}
  \dot B=-\eta\,\widehat G_B\,(\widehat G_B^\top\widehat G_B)^{-1/2}\,S_A^{-1/2},
  \qquad \widehat G_B:=G_B S_A^{-1/2}
\end{equation*}
($\widehat G_B$ is the factor gradient $G_B$ whitened on the right by the other factor's Gram root $S_A^{-1/2}$).
With no accumulation the middle factor is a polar map by \eqref{eq:polar-id}, so
\begin{equation*}
  \dot B=-\eta\,\polar(\widehat G_B)\,S_A^{-1/2}
  =-\eta\,\polar\!\big(G_B S_A^{-1/2}\big)\,S_A^{-1/2},
\end{equation*}
the $B$-update of \Cref{prop:imuon}; the $A$-update is the mirror.

\paragraph{iMuon, momentum-free.} iMuon uses our orientation ($X=BA$). Its fixed-rank
intrinsic spectral LMO (their Cor.~4.1) maps any ambient matrix
$H\in\R^{d_{\mathrm{out}}\times d_{\mathrm{in}}}$ to the factor update
\begin{equation}
  \dot B(H)=\polar\!\big(HA^\top S_A^{-1/2}\big)S_A^{-1/2},\qquad
  \dot A(H)=S_B^{-1/2}\polar\!\big(S_B^{-1/2}B^\top H\big).
  \label{eq:imuon-lmo-ambient}
\end{equation}
The base algorithm (Alg.~1, no momentum) feeds the raw gradient $H=G$, so $HA^\top=G_B$ and
$B^\top H=G_A$ --- and \eqref{eq:imuon-lmo-ambient} is \Cref{prop:imuon} verbatim.

\paragraph{iMuon, with momentum.} The shipped momentum form (their Sec.~K) does \emph{not}
feed the gradient to \eqref{eq:imuon-lmo-ambient}. It first combines the two factor
look-aheads $\tilde G_A=G_A+\beta m_A$ and $\tilde G_B=G_B+\beta m_B$ (buffers $m_A,m_B$,
decay $\beta$) into a single ambient matrix, then feeds that:
\begin{equation*}
  \widehat M=\tilde G_B A+B\tilde G_A .
\end{equation*}
Turning momentum off ($\beta{=}0$) removes the buffers but not the recombination:
\begin{equation*}
  \widehat M=G_B A+B G_A=G\,A^\top A+BB^\top G\ \neq\ G .
\end{equation*}
Feeding $\widehat M$ through \eqref{eq:imuon-lmo-ambient} (using $G_A A^\top=B^\top G_B$),
\begin{equation*}
  \widehat M A^\top=G_B S_A+(BB^\top)G_B
  \quad\Longrightarrow\quad
  \dot B(\widehat M)=\polar\!\big(G_B S_A^{1/2}+(BB^\top)G_B S_A^{-1/2}\big)S_A^{-1/2},
\end{equation*}
which carries an extra term $(BB^\top)G_B$ that the decoupled factor step
$\dot B=\polar(G_B S_A^{-1/2})S_A^{-1/2}$ of \Cref{prop:imuon} does not ($\dot A$ is symmetric).
So switching momentum off does not recover \Cref{prop:imuon}; that needs the recombination
dropped as well, leaving the momentum-free base algorithm. \Cref{prop:imuon} is the
memoryless limit of iMuon's \emph{theory} (Cor.~4.1), not of its shipped optimizer.

\section{Diagonal--Kronecker second-moment estimator for the curvature diagonals}
\label{app:matnormal}
This derives the estimator \eqref{eq:ours-klcoupling} and its coupling
\eqref{eq:ours-closure} from the second-moment model \eqref{eq:ours-sigma} alone --- no
distributional assumption on $G$.

\paragraph{Model and moment identities.} Posit that the gradient second moment is
diagonal--Kronecker, $\Sigma=\E[\mathrm{vec}(G)\mathrm{vec}(G)^\top]\approx Q\otimes P$,
equivalently $\E[G_{ij}G_{kl}]=P_{ik}Q_{jl}$. The only consequence we use is the pair of
moment identities, obtained by contracting that entrywise form: for any
$M\in\mathbb{R}^{d_{\mathrm{in}}\times d_{\mathrm{in}}}$ and
$N\in\mathbb{R}^{d_{\mathrm{out}}\times d_{\mathrm{out}}}$,
\begin{equation}
  \E[G\,M\,G^\top]=\operatorname{tr}(MQ)\,P,\qquad
  \E[G^\top N\,G]=\operatorname{tr}(NP)\,Q.
  \label{eq:moment-id}
\end{equation}
$P,Q$ are fixed only up to scale, $(P,Q)\mapsto(cP,c^{-1}Q)$, which the norm
\eqref{eq:ours-metric} and the operator-norm rescale absorb.

\paragraph{Coupling: covariance of a linear image.} Each factor gradient is a linear
image of $G$, so its row second moment is the partner side conjugated by the factor:
\begin{equation*}
  \mathrm{Cov}\,\mathrm{vec}(G_A)=(I\otimes B^\top)(Q\otimes P)(I\otimes B)
  =Q\otimes\underbrace{B^\top P B}_{C_A},\qquad C_B=A\,Q\,A^\top,
\end{equation*}
which is \eqref{eq:ours-closure}. The partner $C_A$ carries the current $P$ and $C_B$ the
current $Q$; that shared dependence is the coupling between the sides.

\paragraph{Estimator: method of moments.} Put $N=B\,C_A^{-1}B^\top$ in the second
identity of \eqref{eq:moment-id} and use the trace collapse
$\operatorname{tr}(B C_A^{-1}B^\top P)=\operatorname{tr}(C_A^{-1}B^\top P B)=\operatorname{tr}(C_A^{-1}C_A)=r$:
\begin{equation*}
  \E\!\big[\tfrac1r\,G_A^\top C_A^{-1}G_A\big]
  =\E\!\big[\tfrac1r\,G^\top(B C_A^{-1}B^\top)G\big]
  =\tfrac1r\operatorname{tr}(B C_A^{-1}B^\top P)\,Q=Q,
\end{equation*}
so $\tfrac1r\,G_A^\top C_A^{-1}G_A$ is unbiased for $Q$, and its diagonal is the $q_j$ of
\eqref{eq:ours-klcoupling}. The mirror with $M=A^\top C_B^{-1}A$ and
$\operatorname{tr}(A^\top C_B^{-1}A\,Q)=\operatorname{tr}(C_B^{-1}C_B)=r$ gives
$\E[\tfrac1r\,G_B C_B^{-1}G_B^\top]=P$. Setting $C_A=C_B=I$ decouples the sides into two
independent Adafactor diagonals \cite{adafactor}.

\paragraph{Relation to KL-Shampoo.} The alternation --- each diagonal estimated under the
partner's current value, one EMA step per iteration --- is the coupled two-sided
estimator of \cite{klshampoo}, run on the factor gradients $G_A,G_B$ rather than the
dense gradient (their Table~2 lists this diagonal--Kronecker form as moment matching).

\section{Spectral-norm estimation}
\label{app:smax}
\Cref{alg:smax} estimates every operator norm $\norm{\cdot}_2$ in \Cref{alg:ours}. The floor at the larger row/column $\ell_2$ norm (both lower-bound $\smax$) guards against under-estimation: a cold or stale power iteration can under-estimate $\smax$, and an under-estimate at the rescale $W/\norm{W}_2$ inflates the step past $\rho$ and can diverge.

\begin{algorithm}[h]
\caption{Guarded spectral-norm estimate of $\norm{M}_2$}
\label{alg:smax}
\begin{algorithmic}[1]
\Require matrix $M$; warm-start vector $v$ (cached across steps); iterations $K$
\Statex

\State $\ell\gets\max\!\big(\max_i\norm{M_{i,:}}_2,\ \max_j\norm{M_{:,j}}_2\big)$
\Statex

\For{$k=1,\dots,K$} \Comment{power iteration on $M^\top M$}
  \State $w\gets Mv$
  \State $v\gets M^\top w \,/\, \norm{M^\top w}_2$
\EndFor
\Statex

\State \Return $\max\!\big(\norm{Mv}_2,\ \ell\big)$ \Comment{$\ell$ floor}
\end{algorithmic}
\end{algorithm}

\section{Matrix-sign and inverse-sqrt via Gram Newton--Schulz}
\label{app:gramns}
Both spectral primitives of \Cref{alg:ours} --- the matrix sign $\polar(\cdot)$ (the
orthogonal polar factor, \Cref{sec:method}) and the curvature inverse-sqrt $C^{-1/2}$ ---
run as one degree-$5$ Newton--Schulz iteration on the $r\times r$ Gram matrix, sharing a
single coefficient family and using no eigendecomposition (\Cref{alg:gramns}).

\paragraph{Coefficients.} Each iteration applies the odd quintic
$p_t(x)=a_t x+b_t x^3+c_t x^5$, the minimax polynomial that pushes the singular values
toward $1$; composing the $p_t$ carries the whole spectrum to $1$. These are the
PolarExpress coefficients \cite{polarexpress}, computed once at startup and reused at
every step and call site.

\paragraph{Gram iteration.} Since $r\ll d$, both maps iterate on the $r\times r$ Gram
$R$ rather than the rectangular factor \cite{gram_ns} --- about $d/r$ cheaper. With
accumulator $Q$, each step applies
\begin{equation*}
  M_t=a_t I+b_t R+c_t R^2,\qquad Q\gets M_t Q,\qquad R\gets M_t R\,M_t .
\end{equation*}
Both maps run this same iteration, and the accumulator $Q$ is the product of the $M_t$;
they differ only in the seed $R_0$ and what is returned:
\begin{itemize}[leftmargin=1.5em,itemsep=1pt,topsep=2pt]
  \item \textbf{Matrix sign} $\polar(X)$ --- seed $R$ from the Gram of the normalized input, return $Q\tilde X$.
  \item \textbf{Inverse-sqrt} $C^{-1/2}$ --- seed $R$ from the normalized, damped curvature (\Cref{app:damping}), return a rescaled $Q$.
\end{itemize}

\paragraph{Stability and precision.} The coefficients grow quickly across iterations,
so a spurious negative eigenvalue of $R$ can blow up; a restart at iteration $\tau$
re-anchors $R$ from the current iterate \cite{gram_ns}. The iteration runs in fp32 ---
bf16 diverges near the damping floor --- which is cheap on $r\times r$ matrices.
Production uses $K{=}8$ iterations.

\begin{algorithm}[h]
\caption{Gram Newton--Schulz (shared by $\polar$ and the inverse-sqrt)}
\label{alg:gramns}
\begin{algorithmic}[1]
\Require Gram/curvature $R_0$ (SPD, $r\times r$); coefficients $\{(a_t,b_t,c_t)\}_{t=1}^{K}$; restart $\tau$
\State $R\gets R_0$;\quad $Q\gets I$
\For{$t=1,\dots,K$}
  \If{$t=\tau{+}1$} \State $R_0\gets Q R_0 Q^\top$;\quad $R\gets R_0$;\quad $Q\gets I$ \Comment{re-anchor} \EndIf
  \State $M\gets a_t I+b_t R+c_t R^2$
  \State $Q\gets M Q$;\quad $R\gets M R M$
\EndFor
\State \Return $Q$ \Comment{$\to$ polar factor (post-multiply $\tilde X$), or inverse-sqrt (post-scale)}
\end{algorithmic}
\end{algorithm}

\section{Relative damping}
\label{app:damping}
Each curvature inverse-sqrt in \Cref{alg:ours} is relative-damped: for a curvature factor $X$ with top eigenvalue $\lambda_{\max}$,
\begin{equation*}
  X^{-1/2}\ \longmapsto\ \big(X/\lambda_{\max}+\delta I\big)^{-1/2},\qquad \delta>0 .
\end{equation*}
The normalization and the floor each follow from the step's structure.
\begin{itemize}[leftmargin=1.5em,itemsep=2pt,topsep=2pt]
  \item \emph{Normalize by $\lambda_{\max}$.} The whitened polar depends only on the curvature's \emph{shape}, not its scale: $\polar\!\big(C_A^{-1/2}\hat M_A Q^{-1/2}\big)$ is invariant under $X\mapsto cX$, and the operator-norm rescale $W_A/\norm{W_A}_2$ reabsorbs any global factor. The scale-free representative is the normalized spectrum $X/\lambda_{\max}\in(0,1]$.
  \item \emph{Floor by $\delta$.} On that spectrum $\delta$ is one conditioning knob: it caps the metric's condition number at ${\sim}1/\delta$, so a near-null curvature direction cannot amplify the step without bound.
  \item \emph{Warmup.} A still-zero factor ($\lambda_{\max}=0$) maps to $I$, so steps before the EMAs warm up are plain momentum.
\end{itemize}

\section{FLOP overhead of the optimizer}
\label{app:flops}
The wall measurement (\Cref{sec:exp-walltime}) is hardware-specific; the FLOP count is
its architecture-only counterpart, and shows the same near-free result.

\paragraph{Cost per step.} With the base weights frozen, a LoRA step skips the
base-parameter-gradient pass, so the forward/backward costs four matmul-equivalents per
parameter per token (versus six for full fine-tuning):
\begin{equation}
  C_{\mathrm{fb}}=4NT,
  \label{eq:flop-fb}
\end{equation}
with $N$ the (base $+$ adapter) parameter count and $T$ the tokens per step. The
optimizer touches only the adapter and is independent of $T$. Per pair its step
(\Cref{alg:ours,alg:gramns}) is a fixed set of matmuls of two shapes:
\begin{itemize}[leftmargin=1.5em,itemsep=1pt,topsep=2pt]
  \item \emph{$r\times d$ matmuls --- six per factor}, each costing $2r^2d$: the partner
        Gram, the momentum whiten, the polar's Gram and its application, the un-whiten,
        and the flip-flop curvature solve.
  \item \emph{$r\times r$ matmuls}: the two polars and two inverse-sqrts, each a
        $K$-step Gram--Newton--Schulz whose iteration is four $r\times r$ products
        ($R^2$, $MQ$, $MR$, $(MR)M$), so $8r^3$ per iteration.
\end{itemize}
Summing both factors gives the leading-order cost
\begin{equation}
  C_{\mathrm{opt}}=\sum_{\text{pairs}}\Big[
    \underbrace{12\,r^2(d_{\mathrm{in}}{+}d_{\mathrm{out}})}_{6\ \text{matmuls/side}\,\times\,2}
    +\underbrace{32\,K\,r^3}_{4\ \text{calls}\,\times\,8Kr^3}\Big],
  \label{eq:flop-opt}
\end{equation}
every term $r\times r$ or $r\times d$ ($r\ll d$), eigh-free. $C_{\mathrm{opt}}$ is a
fixed per-step cost, so the fraction $C_{\mathrm{opt}}/C_{\mathrm{fb}}\propto1/T$ shrinks
as the batch grows: at the production setting ($r{=}256$, $T{=}32768$) it is under $1\%$
on every base model, far below the measured ${\le}5\%$ wall (\Cref{sec:exp-walltime}).
The optimizer's overhead is thus latency-bound on its many small $r\times r$ kernels, not
FLOP-bound; the FLOP fraction is the floor the wall approaches as the batch grows.

\section{Gauge sensitivity and factor self-balancing}
\label{app:gauge}
The radius is not gauge-invariant: under $(A,B)\mapsto(cA,c^{-1}B)$, $\norm{A}_2{+}\norm{B}_2$ is minimized (largest step) at $\norm{A}_2{=}\norm{B}_2$, so an imbalanced factorization changes the trajectory --- the concern raised for Spectron-style radii \cite{loramuon}.
\begin{itemize}[leftmargin=1.5em,itemsep=1pt,topsep=2pt]
  \item \emph{Self-balancing.} Under \methodname{} the factors balance: $\norm{B}_2/\norm{A}_2$ rises from the $B{=}0$ start toward $1$ and holds (\Cref{fig:gauge}), so the radius runs near its balanced optimum throughout training.
  \item \emph{Tightens with rank.} The balance is closest to $1$ at high rank ($r{=}256$) and looser at low rank ($r{=}32$).
\end{itemize}
