
\section{Estimating the Preconditioners}
\label{app:matnormal-06}
In \Cref{sec:sample-loss-lmo} we treated the preconditioners $P$ and $Q$ as given when writing the update~\eqref{eq:polorafull-update}. Here we describe how we compute them using online updates from the factor gradients and a normalization that makes these updates invariant to the rescaling $(P,Q)\mapsto(aP,a^{-1}Q)$, arriving at the $p,q$ updates~\eqref{eq:ours-klcoupling-06} of \Cref{alg:ours}.

\paragraph{Notation.} Consider a given linear layer. Let $t$ denote the current optimizer step. Whenever it is clear from context, we suppress the subscript $t$. For each step $s\le t$,
let $A_s,B_s$ be the LoRA factors and
$G_s\in\R^{d_{\mathrm{out}}\times d_{\mathrm{in}}}$ the gradient of the batch loss with respect to the weight matrix $W_s=W_0+B_sA_s$. Write
\[
  G_{A,s}=B_s^\top G_s\in\R^{r\times d_{\mathrm{in}}},\qquad
  G_{B,s}=G_sA_s^\top\in\R^{d_{\mathrm{out}}\times r}
\]
for the factor
gradients~\eqref{e-grad-wrt-factors}. 

\paragraph{EMA averaging.} For any sequence $x_s$, write the EMA of the sequence up to step $t$ as
\[
  \EMA^{\beta}_{s\le t}[x_s]:=(1-\beta)\sum_{s\le t}\beta^{t-s} x_s,
\]
with decay factor $\beta$. Just as we replace the factor gradients $G_A$ and $G_B$ in the objective of \Cref{sec:sample-loss-lmo} with averaged versions $\widehat M_A$ and $\widehat M_B$ (line~\ref{ln:mom} of \Cref{alg:ours}), we replace the sample-gradient second-moment matrix $\Sigma$ (\Cref{lem:gradprop}) with the EMA of the outer products of the vectorized full gradients,
\[
  \overline \Sigma_t:=\EMA^{\beta_2}_{s\le t}[\vec(G_s) \vec(G_s)^\top],
\]
with $\beta_2$ the curvature decay of \Cref{alg:ours}.

\paragraph{Factor-gradient moments.}
As in \Cref{sec:sample-loss-lmo}, to make the update tractable we approximate $\overline \Sigma_t \approx Q_t \otimes P_t$~\eqref{eq:pq-factor} with diagonal matrices $P_t = \diag(p_t)$ and $Q_t = \diag(q_t)$. Note, however, that backpropagation gives us efficient access only to the factor gradients, not the full gradients $G_s$. Therefore the natural objects to consider are the corresponding averaged factor-gradient second moments,
\[
  \overline \Sigma_A:=\EMA^{\beta_2}_{s\le t}\bigl[\vec(G_{A,s})\vec(G_{A,s})^\top\bigr],
  \qquad
  \overline \Sigma_B:=\EMA^{\beta_2}_{s\le t}\bigl[\vec(G_{B,s}^\top)\vec(G_{B,s}^\top)^\top\bigr].
\]
If the Kronecker approximation is exact and we replace $A_s$ and $B_s$ with their current values $A$ and $B$, which is reasonable if $A_s \approx A$ and $B_s \approx B$ over recent steps, then
\begin{equation}\label{eq:observable-moments-06}
  \overline \Sigma_A
  = Q\otimes C_A(P),\quad
  \overline \Sigma_B
  = P\otimes C_B(Q),
\end{equation}
where $C_A(P)=B^\top P B$ and $C_B(Q)=A Q A^\top$.

\paragraph{Fitting the preconditioners.} Fitting the observed moments
to the predictions of the Kronecker model
in~\eqref{eq:observable-moments-06} requires a
measure of discrepancy between positive definite matrices. Following KL-Shampoo~\cite{klshampoo}, we use the
log-determinant divergence~\cite{logdetdiv}
\[
  D(\Sigma,S)=\Tr(\Sigma S^{-1})-\log\det(\Sigma S^{-1})-d,
\]
defined for positive definite $d\times d$ matrices, and fit each vector
with the other held fixed:
\begin{equation}\label{eq:factor-fit-06}
  \hat q(P):=\argmin_{q>0}
    D\bigl(\overline \Sigma_A,\diag(q)\otimes C_A(P)\bigr),
  \qquad
  \hat p(Q):=\argmin_{p>0}
    D\bigl(\overline \Sigma_B,\diag(p)\otimes C_B(Q)\bigr).
\end{equation}
A self-consistent pair $(p^*,q^*)$ satisfies $q^*=\hat q(\diag(p^*))$ and $p^*=\hat p(\diag(q^*))$. With $P$ fixed, the first objective in~\eqref{eq:factor-fit-06} separates over the coordinates of $q$. Minimizing each coordinate yields an analytical solution for $\hat q(P)$, and likewise for $\hat p(Q)$:
\begin{equation}\label{eq:conditional-minimizers-06}
\begin{aligned}
  \hat q(P)&=\frac1r\diag\left(\EMA^{\beta_2}_{s\le t}\bigl[G_{A,s}^\top C_A(P)^{-1}G_{A,s}\bigr]\right),\\
  \hat p(Q)&=\frac1r\diag\left(\EMA^{\beta_2}_{s\le t}\bigl[G_{B,s} C_B(Q)^{-1}G_{B,s}^\top\bigr]\right).
\end{aligned}
\end{equation}
Each minimizer depends on the other preconditioner through
$C_A(P)$ or $C_B(Q)$.

\paragraph{Online update.} Re-evaluating the minimizers
in~\eqref{eq:conditional-minimizers-06} at each step would require
storing every past factor gradient, since $C_A(P)$ and $C_B(Q)$ change
as $P$ and $Q$ are updated. \methodname{} instead evaluates each factor
gradient once, at its own step, and accumulates the results:
\begin{equation}\label{eq:ours-klcoupling-06}
\begin{alignedat}{2}
  \hat{q}_t(P_t)
    &= \frac1r\diag\bigl(G_{A,t}^\top C_A(P_t)^{-1}G_{A,t}\bigr),
  &\qquad
  \hat{p}_t(Q_t)
    &= \frac1r\diag\bigl(G_{B,t} C_B(Q_t)^{-1}G_{B,t}^\top\bigr),
  \\
  q_{t+1}
    &= \beta_2q_t+(1-\beta_2)\hat{q}_t(P_t),
  &\qquad
  p_{t+1}
    &= \beta_2p_t+(1-\beta_2)\hat{p}_t(Q_t),
  \\
  Q_{t+1}
    &= \diag\bigl(q_{t+1}/\norm{q_{t+1}}_\infty\bigr),
  &\qquad
  P_{t+1}
    &= \diag\bigl(p_{t+1}/\norm{p_{t+1}}_\infty\bigr).
\end{alignedat}
\end{equation}
Up to the exponentially decayed initialization $\beta_2^{t+1}q_0$, unrolling the
$q_{t+1}$ update in~\eqref{eq:ours-klcoupling-06} gives the
minimizer $\hat q(P)$ of~\eqref{eq:conditional-minimizers-06} with the
step-$s$ matrix $C_A(P_s)$ in place of a fixed $C_A(P)$, and likewise for $p_{t+1}$. In \Cref{alg:ours}, the preconditioner updates (lines~\ref{ln:qmom}
and~\ref{ln:pmom}) reuse the matrices $C_A$ and $C_B$ already computed for the direction step (line~\ref{ln:pqnorm}), so each step inverts $C_A$ and $C_B$ once rather than twice.

\paragraph{Normalization.} We now explain why we normalize $q_{t+1},p_{t+1}$
in~\eqref{eq:ours-klcoupling-06}. The pair $(P, Q)$ enters the
approximation of $\overline \Sigma_t$ only through the product
$Q \otimes P$, and
\[
  (a^{-1}Q)\otimes(aP)=Q\otimes P
  \qquad\text{for every } a>0,
\]
so we want the optimizer to behave identically at $(P,Q)$ and
$(aP,a^{-1}Q)$. The update direction is the same at both, but
\[
  \hat q_t(aP_t)=a^{-1}\hat q_t(P_t),
  \qquad
  \hat p_t(a^{-1}Q_t)=a\,\hat p_t(Q_t),
\]
so we normalize $q_{t+1},p_{t+1}$ in~\eqref{eq:ours-klcoupling-06} to
make the $p,q$ updates invariant as well.
