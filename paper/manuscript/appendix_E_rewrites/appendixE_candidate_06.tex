
\section{Estimating the Curvature Factors}
\label{app:matnormal-06}
\Cref{sec:sample-loss-lmo} approximates the per-sample second moment
$\Sigma$ by a Kronecker product $Q\otimes P$ of curvature
factors~\eqref{eq:pq-factor} and treats the factors as given. This
appendix estimates them from the factor gradients of recent batches,
which backpropagation already provides.
\Cref{sec:weighted-estimation-06} obtains the $p,q$ recursion of
lines~\ref{ln:qmom} and~\ref{ln:pmom}, and \Cref{sec:scale-placement-06}
explains why the accumulated vectors are normalized before they are reused.

\subsection{Estimation from the Factor Gradients}
\label{sec:weighted-estimation-06}

\paragraph{Notation.} Let $s$ index optimizer steps, $t$ the current
step, and $G_s$ the gradient of the step-$s$ batch loss with respect to
the weight matrix $W_s$, with $g_s=\vec(G_s)$.
Backpropagation provides the factor gradients~\eqref{e-grad-wrt-factors},
\[
  G_{A,s}=B_s^\top G_s,
  \qquad
  G_{B,s}=G_sA_s^\top .
\]
The derivation below uses two Kronecker-product identities:
\begin{itemize}[leftmargin=1.5em,itemsep=1pt,topsep=2pt]
  \item the \emph{vectorization identity},
    $\vec(XYZ^\top)=(Z\otimes X)\vec(Y)$;
  \item the \emph{mixed-product rule},
    $(X_1\otimes Y_1)(X_2\otimes Y_2)=X_1X_2\otimes Y_1Y_2$.
\end{itemize}

\paragraph{Batch-gradient averaging.} The moment $\Sigma$
in~\eqref{eq:pq-factor} averages outer products of per-sample gradients
(\Cref{lem:gradprop}), but backpropagation returns only their batch
average $G_s$. \methodname{} therefore replaces the per-sample outer
products with batch outer products $g_sg_s^\top$, averaged over recent
steps. For any sequence $d_s$, write
\[
  \EMA_{s\le t}[d_s]:=(1-\beta_2)\sum_{s\le t}\beta_2^{t-s}d_s,
\]
a weighted sum of the realized past values, with $\beta_2$ the curvature
decay of \Cref{alg:ours}. The moment the Kronecker model approximates is
then $\Sigma_t:=\EMA_{s\le t}[g_sg_s^\top]$.

\paragraph{Factor-gradient moments under the Kronecker model.}
\Cref{alg:ours} keeps the curvature factors diagonal, so we fit them in
the form
\[
  P=\diag(p),\qquad Q=\diag(q),\qquad
  C_A(p)=B^\top P B,\qquad C_B(q)=AQA^\top ,
\]
where $p,q$ range over positive vectors and $A,B$ denote the current
LoRA factors $A_t,B_t$. Let
\[
  \Sigma_A:=\EMA_{s\le t}\bigl[\vec(G_{A,s})\vec(G_{A,s})^\top\bigr],
  \qquad
  \Sigma_B:=\EMA_{s\le t}\bigl[\vec(G_{B,s}^\top)\vec(G_{B,s}^\top)^\top\bigr]
\]
denote the averaged moments of the factor gradients.
Suppose temporarily that the Kronecker model~\eqref{eq:pq-factor} holds
exactly for the averaged moment, $\Sigma_t=Q\otimes P$. The vectorization
identity gives $\vec(G_{A,s})=(I\otimes B_s^\top)g_s$ and
$\vec(G_{B,s}^\top)=(I\otimes A_s)\vec(G_s^\top)$. Replacing $A_s,B_s$
inside the average by the current $A,B$, the mixed-product rule then
gives
\begin{equation}\label{eq:observable-moments-06}
\begin{aligned}
  \Sigma_A
  &\approx(I\otimes B^\top)(Q\otimes P)(I\otimes B)
    =\diag(q)\otimes C_A(p),\\
  \Sigma_B
  &\approx(I\otimes A)(P\otimes Q)(I\otimes A^\top)
    =\diag(p)\otimes C_B(q),
\end{aligned}
\end{equation}
where the second line uses that transposing $G_s$ exchanges the
Kronecker factors, so the averaged moment of $\vec(G_s^\top)$ is
$P\otimes Q$.
In each moment, one unknown vector appears as the diagonal Kronecker
factor while the other enters only through an $r\times r$ matrix. We
therefore estimate $q$ and $p$ by fitting the observed moments to the
forms in~\eqref{eq:observable-moments-06}, without assuming that the
model holds exactly.

\paragraph{Log-determinant fits.} Following
KL-Shampoo~\cite{klshampoo}, let
\[
  D(\Sigma,S)=\Tr(\Sigma S^{-1})-\log\det(\Sigma S^{-1})-d
\]
be the log-determinant divergence~\cite{logdetdiv} between positive definite
$d\times d$ matrices. For a fixed $p$ we fit $q$ to $\Sigma_A$, and for
a fixed $q$ we fit $p$ to $\Sigma_B$:
\begin{equation}\label{eq:factor-fit-06}
  \hat q(p):=\argmin_{q>0}
    D\bigl(\Sigma_A,\diag(q)\otimes C_A(p)\bigr),
  \qquad
  \hat p(q):=\argmin_{p>0}
    D\bigl(\Sigma_B,\diag(p)\otimes C_B(q)\bigr).
\end{equation}

\paragraph{Closed-form minimizers.} With $p$ fixed, the first objective
in~\eqref{eq:factor-fit-06} separates over the coordinates of $q$, since
the fitted matrix has inverse $\diag(q)^{-1}\otimes C_A(p)^{-1}$, the
vectorization identity evaluates the trace term on the factor gradient,
and the determinant of the Kronecker product factorizes. Up to additive
terms that do not depend on $q$,
\[
  D\bigl(\Sigma_A,\diag(q)\otimes C_A(p)\bigr)
  =\sum_{j=1}^{d_{\mathrm{in}}}
  \left[
    \frac{\EMA_{s\le t}\bigl[(G_{A,s}^\top C_A(p)^{-1}G_{A,s})_{jj}\bigr]}{q_j}+r\log q_j
  \right].
\]
Minimizing each coordinate, and applying the same computation to
$\Sigma_B$, gives
\begin{equation}\label{eq:conditional-minimizers-06}
  \hat q(p)=\frac1r\diag\left(\EMA_{s\le t}\bigl[G_{A,s}^\top C_A(p)^{-1}G_{A,s}\bigr]\right),
  \qquad
  \hat p(q)=\frac1r\diag\left(\EMA_{s\le t}\bigl[G_{B,s} C_B(q)^{-1}G_{B,s}^\top\bigr]\right).
\end{equation}
Each fit still depends on the other factor through $C_A(p)$ or $C_B(q)$.
A self-consistent pair satisfies $q^*=\hat q(p^*)$ and $p^*=\hat p(q^*)$,
and \methodname{} approaches one by evaluating each fit at the current
estimate of the other factor.

\paragraph{Online recurrence.} Evaluating the fits
in~\eqref{eq:conditional-minimizers-06} anew at each step would require
storing every past factor gradient, since $C_A(p)$ and $C_B(q)$ change
as $p$ and $q$ are updated. \methodname{} instead evaluates each factor
gradient once, at its own step, with the matrices
$C_{A,t}=B_t^\top P_tB_t$ and $C_{B,t}=A_tQ_tA_t^\top$ formed from the
normalized factors $P_t,Q_t$, and accumulates the results:
\begin{equation}\label{eq:ours-klcoupling-06}
\begin{alignedat}{2}
  \hat{q}_t(P_t)
    &= \frac1r\diag\bigl(G_{A,t}^\top C_{A,t}^{-1}G_{A,t}\bigr),
  &\qquad
  \hat{p}_t(Q_t)
    &= \frac1r\diag\bigl(G_{B,t} C_{B,t}^{-1}G_{B,t}^\top\bigr),
  \\
  q_{t+1}
    &= \beta_2q_t+(1-\beta_2)\hat{q}_t(P_t),
  &\qquad
  p_{t+1}
    &= \beta_2p_t+(1-\beta_2)\hat{p}_t(Q_t),
  \\
  Q_{t+1}
    &= \diag\bigl(q_{t+1}/\norm{q_{t+1}}_\infty\bigr),
  &\qquad
  P_{t+1}
    &= \diag\bigl(p_{t+1}/\norm{p_{t+1}}_\infty\bigr).
\end{alignedat}
\end{equation}
Up to the decayed initialization $\beta_2^{t+1}q_0$, unrolling the
recursion gives $q_{t+1}=\EMA_{s\le t}[\hat q_s(P_s)]$, which is the fit
$\hat q(p)$ of~\eqref{eq:conditional-minimizers-06} with the step-$s$
matrix $C_{A,s}$ in place of a fixed $C_A(p)$, and likewise for
$p_{t+1}$.
The optimizer retains only $p_t$ and $q_t$, and the inverses in the
implemented optimizer use the relative damping of
\Cref{app:damping}. Within a step,
the matrices $C_{A,t},C_{B,t}$ are formed once and used twice, first in
the direction and magnitude update~\eqref{eq:polora-update} and then in
the increments $\hat q_t,\hat p_t$, so the increment from step $t$ first
affects the direction at step $t+1$.

\subsection{Scale Normalization}
\label{sec:scale-placement-06}
\methodname{} normalizes the accumulated vectors $q_{t+1},p_{t+1}$ when
forming $Q_{t+1},P_{t+1}$ in~\eqref{eq:ours-klcoupling-06}. To see why,
let $D_A(P,Q)$ and $D_B(P,Q)$ denote the direction matrices
in~\eqref{eq:factor-directions}, and multiply the stored factors by any
$a,b>0$, $(P_t,Q_t)\mapsto(aP_t,bQ_t)$. The factor update at step $t$
does not change, but the increments at step $t+1$ shrink by $a^{-1}$ and
$b^{-1}$.

\paragraph{The factor update is unchanged.} The rescaling multiplies
$C_A=B^\top PB$ by $a$ and $C_B=AQA^\top$ by $b$, and the matrix sign
ignores positive scalars, so
\[
  D_A(aP_t,bQ_t)=(ab)^{-1/2}D_A(P_t,Q_t),
  \qquad
  D_B(aP_t,bQ_t)=(ab)^{-1/2}D_B(P_t,Q_t).
\]
The update~\eqref{eq:polora-update} applies $D_A/\norm{D_A}_2$ and
$D_B/\norm{D_B}_2$ at the update size~\eqref{eq:ours-rho}, which contains
neither $P$ nor $Q$, so the factor $(ab)^{-1/2}$ cancels in the division.

\paragraph{The next increments shrink.} Because $C_A(p)$ is linear in
$p$ and $C_B(q)$ is linear in $q$, the increments
in~\eqref{eq:ours-klcoupling-06} obey
\[
  \hat q_{t+1}(aP_{t+1})=a^{-1}\hat q_{t+1}(P_{t+1}),
  \qquad
  \hat p_{t+1}(bQ_{t+1})=b^{-1}\hat p_{t+1}(Q_{t+1}).
\]
The multipliers $a,b$ are therefore invisible at step $t$ yet change
every later increment. Since $ap/\norm{ap}_\infty=p/\norm{p}_\infty$ for
every $a>0$, normalizing removes them, and the factors entering each
step have unit largest entry whatever magnitudes the EMA accumulates.
